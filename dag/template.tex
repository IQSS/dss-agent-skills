% ---- node + edge styles ----
% Node roles:  exposure | outcome | msas | mediator | obs | latent
%   msas     = minimal sufficient adjustment set (measured confounders to
%              condition on for the exposure->outcome effect; dagitty term).
%   mediator = on a directed exposure->outcome path; do NOT adjust.
%   obs      = measured, present in the graph, neither confounder nor mediator.
%   latent   = unmeasured / unobserved.
% Edge classes: causal | conf | latentedge | other
%   causal     = lies on a directed exposure->outcome path (the effect estimated).
%   conf       = sourced from an msas node (opens a back-door path).
%   latentedge = optional highlight for an edge out of a latent confounder.
%   other      = remaining structural edges.

% ---- legend helpers (emit only the entries that apply to THIS dag) ----
% Prefer the entry wrappers: \lentry{<role>}{<caption>}   \eentry{<color>}{<caption>}
%                            \eentrydash{<color>}{<caption>}   (latent)
% Bare swatches \lnode / \ledge / \ledgedash are available if a caption needs
% custom markup, but then wrap the pair in \mbox yourself (see below).
% Keep swatches small — the legend should read as a compact footer, not a banner.
%
% Swatch-vs-text vertical centring, one knob PER SWATCH TYPE. They deliberately do
% not share a value: the three macros build different objects (a tikz node, a tikz
% path, a plain rule) and do not land alike for a given offset under the global
% font=\sffamily\footnotesize set above, which the nested \tikz swatches inherit.
%
% Check alignment in the completed figure, because inherited font settings and
% the caption's ascenders and descenders affect the apparent vertical centre.
\newcommand{\swnode}{-0.5ex}                       % \lnode      (tikz node)
\newcommand{\swdash}{-0.65ex}                      % \ledgedash  (tikz path)
\newcommand{\swrule}{\dimexpr0.5ex-0.55pt\relax}   % \ledge      (rule lift, from its bottom)
\newcommand{\lnode}[1]{\tikz[baseline=\swnode]\node[#1, minimum width=0.38cm, minimum height=0.24cm, inner sep=1.5pt]{};}
\newcommand{\ledge}[1]{{\color{#1}\rule[\swrule]{0.5cm}{1.1pt}}}
\newcommand{\ledgedash}[1]{\tikz[baseline=\swdash]\draw[#1, line width=1.1pt, dashed](0,0)--(0.5cm,0);}
% Entry wrappers: \mbox keeps a swatch glued to its caption so a legend line can
% only break BETWEEN entries, never inside one. Without this, a legend node with
% `text width` hyphenates mid-phrase ("confound-/ing path").
\newcommand{\lentry}[2]{\mbox{\lnode{#1}\ #2}}
\newcommand{\eentry}[2]{\mbox{\ledge{#1}\ #2}}
\newcommand{\eentrydash}[2]{\mbox{\ledgedash{#1}\ #2}}
